9.3 Row-level security

The browser holds Supabase's public key, so the database itself must decide what each request may see. Row-level security (RLS) is switched on for every table, and each table has policies such as "a farm row is visible only if its user_id is the signed-in user". Logged-out visitors get no access at all.

supabase/setup.sql — examples of security policies (excerpt)
alter table public.farms enable row level security;

-- Directly owned rows
create policy farms_own on public.farms for all to authenticated
  using (user_id = (select auth.uid())) with check (user_id = (select auth.uid()));

-- Rows owned through a parent: a crop is visible only if its farm belongs to the user
create policy crop_cycles_own on public.crop_cycles for all to authenticated
  using (exists (select 1 from public.farms f where f.id = farm_id and f.user_id = (select auth.uid())))
  with check (exists (select 1 from public.farms f where f.id = farm_id and f.user_id = (select auth.uid())));

-- Community: everyone signed in can read; you can only write as yourself, never as the AI
create policy replies_read on public.community_replies for select to authenticated using (true);
create policy replies_insert_own on public.community_replies for insert to authenticated
  with check (user_id = (select auth.uid()) and is_ai = false);

-- Anonymous (logged-out) visitors get nothing.
revoke all on all tables in schema public from anon;

src/lib/rate-limit.ts — sliding-window limiter (excerpt)
export function checkRateLimit(bucket: keyof typeof AI_LIMITS, userId: string) {
  const { limit, windowMs } = AI_LIMITS[bucket]
  const now = Date.now()
  const key = `${bucket}:${userId}`
  const recent = (hits.get(key) ?? []).filter((t) => now - t < windowMs)

  if (recent.length >= limit) {
    hits.set(key, recent)
    return { ok: false, retryAfterSec: Math.max(1, Math.ceil((recent[0] + windowMs - now) / 1000)) }
  }
  recent.push(now)
  hits.set(key, recent)
  return { ok: true, retryAfterSec: 0 }
}

Note: the limiter keeps its counts in the memory of each server instance. This stops accidental or casual overuse; a shared store such as Redis would be the upgrade for large scale.

10.3 Keeping keys secret

Satellite images are stored by Agro Monitoring and need the API key to be viewed. Instead of giving the key to the browser, the browser asks the app's own server for the image; the server checks that the farm belongs to the farmer and that the address is really Agro Monitoring, adds the key and passes the image back.

src/app/api/farms/[id]/satellite/route.ts — image proxy with address check (simplified excerpt)
// Only the owner's farm, and only the image URL Agro itself returned in the last scan.

  .eq('id', id).eq('user_id', user.id).maybeSingle()

// SSRF guard: never fetch anything except Agro Monitoring over HTTPS.

target.protocol = 'https:'

const upstream = await fetch(target, { next: { revalidate: 3600 } })

10.4 Safe errors

src/lib/api-error.ts — one helper used by every route (simplified excerpt)
/**
 * Log the real error on the server and return a safe, generic message to the browser.
 * Database / upstream error text can reveal table names, policies or request URLs
 * (which may contain API keys), so it never leaves the server.
 */
export function serverError(tag: string, err: unknown, message = 'Something went wrong. Please try again.', status = 500) {
  const detail = err instanceof Error ? err.message : String(err)
  console.error(`[${tag}]`, detail)
  return NextResponse.json({ error: message }, { status })
}
