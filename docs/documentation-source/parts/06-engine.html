7.2 The fallback algorithm

A request walks down its chain until one model gives an answer. Models that recently failed are moved to the end of the line (never removed), and a bad or missing key skips the rest of that provider's models.

src/lib/ai.ts — the main loop of aiGenerate (simplified excerpt)
for (const step of orderChain(opts.chain ?? TEXT_CHAIN, opts.exclude)) {
  if (deadProviders.has(step.provider)) continue          // bad key: skip provider
  const remaining = deadline - Date.now()
  if (remaining < 2_000) break                            // 55-second budget used up

  const t = withTimeout(Math.min(opts.attemptTimeoutMs ?? 30_000, remaining))
  try {
    const text = await callOnce(step, req, opts, t.signal)
    attempts.push({ provider: step.provider, model: step.model, ok: true, ms: Date.now() - t0 })
    return { text, provider: step.provider, model: step.model, latencyMs: Date.now() - started, attempts }
  } catch (raw) {
    const err = asProviderError(raw, t.signal)
    applyCooldown(step, err)                                // rest a failing model
    if (err instanceof ProviderError && err.providerFatal) deadProviders.add(step.provider)
    attempts.push({ provider: step.provider, model: step.model, ok: false, status: err.status, error: err.message })
  } finally {
    t.clear()
  }
}
throw new AiChainError('All AI providers failed', attempts)

src/lib/ai.ts — resting and ordering models (excerpt)
function applyCooldown(step: ChainStep, err: unknown) {
  if (!(err instanceof ProviderError) || err.simulated) return
  if (err.status === 429) coolDown(step, err.retryAfterMs ?? 60_000)
  else if (err.status === 404) coolDown(step, 10 * 60_000)    // model retired / renamed
  else if (err.status && err.status >= 500) coolDown(step, 20_000)
  else if (err.status === 408) coolDown(step, 30_000)         // our own timeout
}

function orderChain(chain: ChainStep[], exclude?: Set<string>): ChainStep[] {
  const now = Date.now()
  const usable  = chain.filter((s) => !exclude?.has(stepKey(s)))
  const healthy = usable.filter((s) => (cooldownUntil.get(stepKey(s)) ?? 0) <= now)
  const cooling = usable.filter((s) => (cooldownUntil.get(stepKey(s)) ?? 0) > now)
  return [...healthy, ...cooling]          // resting models go last, never removed
}

7.4 Streaming answers with failover

For the advisor chat, the answer is streamed. The gateway waits for the first word from a model before it commits to that model; a failure before that point moves on to the next model, invisible to the farmer. If a model fails in the middle of an answer, the farmer keeps the words already received and the stream ends cleanly.

Streamed replies arrive as Server-Sent Events. Network packets can cut a line — or a Hindi or Telugu character — in half, so the parser keeps a buffer and decodes text in streaming mode:

src/lib/ai.ts — Server-Sent Events parser (excerpt)
async function* sseData(body: ReadableStream<Uint8Array>): AsyncGenerator<string> {
  const reader = body.getReader()
  const decoder = new TextDecoder()
  let buffer = ''
  while (true) {
    const { done, value } = await reader.read()
    buffer += done ? decoder.decode() : decoder.decode(value, { stream: true })
    let nl: number
    while ((nl = buffer.indexOf('\n')) >= 0) {
      const line = buffer.slice(0, nl).replace(/\r$/, '')
      buffer = buffer.slice(nl + 1)
      if (line.startsWith('data:')) yield line.slice(5).trimStart()
      // lines starting with ':' are keep-alive comments; blank lines end events
    }
    if (done) return
  }
}

7.5 Structured JSON answers

Diagnosis reports and task plans must be machine-readable. aiJson asks the model for JSON (native JSON mode on both providers), extracts the JSON object even if the model adds extra words, checks it with a validation function, and if it is broken, asks a different model.

src/lib/ai.ts — aiJson (simplified excerpt)
export async function aiJson<T>(req: AiRequest, validate: (value: unknown) => T, opts = {}) {
  const exclude = new Set(opts.exclude ?? [])
  for (let i = 0; i <= (opts.maxParseRetries ?? 2); i++) {
    const result = await aiGenerate(req, { ...opts, json: true, exclude })
    try {
      return { ...result, data: validate(extractJsonObject(result.text)) }
    } catch (e) {
      // this model returned invalid JSON: never ask it again for this request
      exclude.add(stepKey({ provider: result.provider, model: result.model }))
    }
  }
  throw new Error('AI returned invalid JSON')
}
